using System.Collections.Generic;
using System.IO;
using System.Threading.Tasks;
using gensys.shared;
using System.Linq;



namespace clientApp.Data
{
    public class SQLiteEFDataStore: gensys.client.shared.IDataStore
    {
  
        /// <summary>
        /// 操作锁
        /// </summary>
        private static object dbLock = new object();


        CliDbContext dbContext;

       
        public SQLiteEFDataStore(CliDbContext dbContext)
        {
            this.dbContext = dbContext;
        }

        public Room CopyFrom(ChatItem room,Room dbRoom=null)
        {
            Room roomCopy = dbRoom?? new Room();
            if (roomCopy.Id != room.Id)
            {
                roomCopy.Id=room.Id;
            }
            roomCopy.UserCount = room.UserCount;
            roomCopy.Creator = room.Creator;
            roomCopy.Name = room.Name;
            roomCopy.Description = room.Description;
            roomCopy.SyncTime = room.SyncTime;
            roomCopy.NoViewCount = room.NoViewCount;
            roomCopy.MsgType = room.MsgType;
            roomCopy.ChatType = room.ChatType;
            roomCopy.ConnectionId = room.ConnectionId;
            roomCopy.LastReadTime = room.LastReadTime;
            roomCopy.Face = room.Face;
            roomCopy.Remark = room.Remark;
            roomCopy.LastUpdateTime = room.LastUpdateTime;
            return roomCopy;

        }

        public  Task AddOrUpdateRoom(ChatItem room)
        {        
           var r= dbContext.Room.FirstOrDefault(m => m.Id == room.Id);
          if (r == null)
            {
                Room copy = CopyFrom(room);
                dbContext.Add(copy);
            }
            else
            {
                dbContext.Attach(r);
                CopyFrom(room,r);               
            }
            dbContext.SaveChanges();
            return Task.CompletedTask;
        }



        public  Task<T> GetConfig<T>(string configId)
        {
            try
            {

                Console.WriteLine($"读取配置{configId}");
                var config=dbContext.CliConfig.FirstOrDefault(m => m.Id == configId);

            
                //config =  _database.GetAsync<CliConfig>(configId).Result;
                if (config != null)
                {
                    var data = System.Text.Json.JsonSerializer.Deserialize<T>(config.Config);
                    return  Task.FromResult(data);
                }

                return Task.FromResult<T>(default(T));
            }catch(Exception ex)
            {
                Console.WriteLine(ex);
                throw ex;
            }
        }



        public  Task SetConfig(string configId, object config)
        {
            CliConfig c = new CliConfig();
            c.Id = configId;
           string strConfig = System.Text.Json.JsonSerializer.Serialize(config);
            var dbConfig=dbContext.CliConfig.FirstOrDefault(m => m.Id == configId);
            if (dbConfig != null)
            {
                dbContext.Attach(dbConfig);
                dbConfig.Config = strConfig;
            }
            else
            {
                dbConfig = new CliConfig();
                dbConfig.Id = configId;
                dbConfig.Config = strConfig;
                dbContext.Add(dbConfig);
            }
            dbContext.SaveChanges();
            return Task.CompletedTask;
        }

        /// <summary>
        /// 移除配置
        /// </summary>
        /// <param name="configId"></param>
        /// <returns></returns>
        public  Task RemoveConfig(string configId)
        {
            var dbConfig = dbContext.CliConfig.FirstOrDefault(m => m.Id == configId);
            if (dbConfig != null)
            {
                dbContext.Remove(dbConfig);
                dbContext.SaveChanges();
            }
            return Task.CompletedTask;
        }

        public async Task<gensys.shared.Message> GetLastMsg(Guid roomId)
        {
            var msg=  dbContext.Message.Where(m => m.ChatRoomId == roomId).OrderByDescending(m => m.SendTime).FirstOrDefault();
            if (msg != null)
            {

                msg.From =await GetRoom(msg.FromId);
                msg.To = await GetRoom(msg.ToId);
            }
            return msg;
 
        }

        public  Task<int> GetMessageCount(Guid roomId, DateTime startTime)
        {
            return Task.FromResult(dbContext.Message.Where(m => m.Id == roomId && m.SendTime > startTime).Count());
        }

      
        public  Task<ChatItem> GetRoom(Guid roomId)
        {
            ChatItem item= dbContext.Room.Where(m => m.Id == roomId).FirstOrDefault();
            return  Task.FromResult(item);
            
        }

        public  Task<List<ChatItem>> GetRooms()
        {
            var list = dbContext.Room.ToList();
            List<gensys.shared.ChatItem> items = new List<ChatItem>();
            foreach (var item in list)
            {
                items.Add(item);
            }
            return Task.FromResult(items);
        }

 

        public async Task SetAllRead(Guid roomId)
        {
            try
            {
                Room dbRoom = dbContext.Room.FirstOrDefault(m=>m.Id==roomId);
                if (dbRoom != null)
                {
                    dbContext.Attach(dbRoom);
                    //获取最近一条消息
                    var lastMsg = await GetLastMsg(dbRoom.Id);
                    if (lastMsg != null)
                    {
                        if (lastMsg.SendTime > dbRoom.LastReadTime)
                        {
                            dbRoom.LastReadTime = lastMsg.SendTime.AddSeconds(1);
                        }
                    }
                    dbRoom.NoViewCount = 0;
                    dbContext.SaveChanges();
                }
            }
            catch { }
        }

        public  Task DeleteRoom(Guid? roomId)
        {
            if (roomId == null) //删除所有房间及消息
            {
                dbContext.Message.RemoveRange(dbContext.Message.ToList());
                dbContext.Room.RemoveRange(dbContext.Room.ToList());
                dbContext.SaveChanges();
            }
            else
            {
                var room = GetRoom(roomId.Value);
                if (room != null)
                {
                    dbContext.Message.RemoveRange(dbContext.Message.Where( m =>m.ChatRoomId==roomId.Value));
                    dbContext.Room.Where(m => m.Id == roomId);
                    dbContext.SaveChanges();
                }
            }
            return Task.CompletedTask;
        }


        public Data.Message CopyFrom(gensys.shared.Message msg)
        {
            Data.Message dbMsg = new Message();
            dbMsg.ToId = msg.To.Id;
            dbMsg.FromId = msg.From.Id;
            dbMsg.Sended= msg.Sended;
            dbMsg.Readed= msg.Readed;
            dbMsg.ChatRoomId = msg.ChatRoomId;
            dbMsg.Content = msg.Content;
            dbMsg.Id= msg.Id;
            dbMsg.SendTime = msg.SendTime;
            dbMsg.Remark = msg.Remark;
            dbMsg.Type = msg.Type;
            return dbMsg;
        }

        public  Task WriteMessages(Guid myId, gensys.shared.Message[] msgs)
        {
            try
            {
                if (msgs == null || msgs.Length == 0) return Task.CompletedTask;
                foreach (var msg in msgs)
                {
                    if (msg != null)
                    {
                        //消息存储时，房间ID永远不能为自己！
                        msg.ChatRoomId = (msg.ChatRoomId == myId ? msg.From.Id : msg.ChatRoomId);
                        var dbMsg= dbContext.Message.FirstOrDefault(m=>m.Id==msg.Id);
                        if (dbMsg != null)
                        {
                            dbContext.Attach(dbMsg);
                            dbMsg.ChatRoomId = msg.ChatRoomId;
                            dbMsg.Content = msg.Content;
                            dbMsg.Sended= msg.Sended;
                            dbMsg.Readed = msg.Readed;
                            dbMsg.SendTime = msg.SendTime;
                            dbMsg.Type = msg.Type;

                        }else
                        {
                            dbContext.Add(CopyFrom(msg));
                        }
                        dbContext.SaveChanges();
                    }
                }

            }
            catch (Exception ex)
            {
                ;
                Console.Error.WriteLine($"writemsg:{ex}");
                Console.Error.WriteLine(msgs)
;
            }
            return Task.CompletedTask;
        }

        public  Task ClearChat(Guid? roomId)
        {
            if (roomId == null)
            {
                dbContext.Message.RemoveRange(dbContext.Message.ToList());
                dbContext.SaveChanges();
            }
            else
            {
                dbContext.Message.RemoveRange(dbContext.Message.Where(m => m.ChatRoomId == roomId));
                dbContext.SaveChanges();
            }
    
            return Task.CompletedTask;
        }

        public async Task<List<gensys.shared.Message>> GetMessages( Guid roomId, DateTime? endTime, int count)
        {
           
            List<Data.Message> msgs = null;
            List<gensys.shared.Message> msgs2 = new List<gensys.shared.Message>();
            if (endTime == null) msgs =  dbContext.Message.Where(m=>m.ChatRoomId==roomId).OrderByDescending(m => m.SendTime).Take(count).ToList();
            else msgs =  dbContext.Message.Where(m =>m.ChatRoomId==roomId &&  m.SendTime < endTime).OrderByDescending(m => m.SendTime).Take(count).ToList();
            SortedList<Guid, ChatItem> user = new SortedList<Guid, ChatItem>();
            foreach (var msg in msgs)
            {

                if (user.TryGetValue(msg.FromId,out var u))
                {
                    msg.From = u;

                }
                else
                {
                    msg.From = await GetRoom(msg.FromId);
                    if (msg.From == null) msg.From = new ChatItem { Id = msg.FromId };
                    user.Add(msg.FromId, msg.From);
                }

                if (user.TryGetValue(msg.ToId, out var _to))
                {
                    msg.To = _to;

                }
                else
                {
                    msg.To = await GetRoom(msg.FromId);
                    if (msg.To == null) msg.To = new ChatItem { Id = msg.ToId };
                    user.Add(msg.ToId, msg.To);
                }
                msgs2.Add(msg);
            }
            return msgs2;
        }
    }
}
